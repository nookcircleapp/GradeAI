       In both `exams.ts` and `student.ts`, add a helper function:
       ```typescript
       const TIMEOUT_MS = 30_000

       async function fetchWithTimeout(url: string, options?: RequestInit): Promise<Response> {
         const controller = new AbortController()
         const timeoutId = setTimeout(() => controller.abort(), TIMEOUT_MS)
         try {
           const response = await fetch(url, { ...options, signal: controller.signal })
           return response
         } catch (error) {
           if (error instanceof DOMException && error.name === 'AbortError') {
             throw new Error('Request timed out. Please check your connection and try again.')
           }
           throw error
         } finally {
           clearTimeout(timeoutId)
         }
       }
       ```

    2. **Replace all `fetch()` calls with `fetchWithTimeout()`** in both files:

       - Keep all existing error handling (response.ok checks, error messages) unchanged
       - The timeout wrapper is transparent — same signature as fetch, just adds AbortController
